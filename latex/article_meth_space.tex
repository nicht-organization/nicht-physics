% ABSTRACT
% ============================================================

\begin{abstract}
This paper establishes a minimal, formal distinction between two modes of
mathematical consistency.
\emph{1-Consistency} denotes standard internal consistency of a formal system
under its specified inference rules ($S \nvdash \bot$). \emph{0-Consistency}
denotes the consistency of a system under arbitrary finite or transfinite
iteration of a relevant self-negation or inversion operator $I$. We formalize
0-Consistency as the 1-consistency of the inductive limit system 
$S_I^\infty = \bigcup_{n \in \mathbb{N}} I^n(S)$ and define the Sufficient 
Invariant Baseline $B_0 \equiv \operatorname{Fix}(I)$ as its structural ground
state. While the logical labels $0$ and $1$ are dual under static truth-value 
assignment, non-closure under $I$ breaks this symmetry, distinguishing local 
rule-compliance from global inversion stability. This framework provides a 
reusable, non-metaphysical foundation for specialized investigations in 
geometry, number theory, and mathematical physics.
\end{abstract}

\section{Introduction}
% ============================================================

Standard mathematical reasoning fixes a formal theory 
$S = (\mathcal{L}, \mathcal{A}, \mathcal{R})$ with language 
$\mathcal{L}$, axioms $\mathcal{A}$, and inference rules 
$\mathcal{R}$, evaluating consistency entirely within 
$\text{Th}(S)$. This operational baseline represents internal
rule-compliance.

However, formal systems frequently generate operations, such 
as - additive inversion, multiplicative duality, or geometric
complementation, whose domain extends beyond the originating
system. A system may remain consistent within its defined
boundaries while failing to sustain closure under arbitrary
iteration of its intrinsic inversion operators.

We define this boundary through the core distinction:
\[
\boxed{
\text{Internal Syntactic Consistency (1-Consistency)}
\quad\neq\quad
\text{Closure Under Iterated Inversion (0-Consistency)}
}
\]
This paper formalizes this distinction as a domain-agnostic
foundation for subsequent applications in analytic number theory,
 algebraic geometry, and field theory.

\section{Formalization of the Two Modes}
% ============================================================

Let $S = (\mathcal{L}, \mathcal{A}, \mathcal{R})$ be a formal system, and let 
$\vdash_S$ denote syntactic provability. We denote logical absurdity by $\bot$.

\subsection{1-Consistency}

\begin{definition}[1-Consistency]
A formal system $S$ is \emph{1-consistent} (denoted $1\text{C}$) if it cannot
derive a contradiction under its specified rules:
\[
S \nvdash_S \bot.
\]
Equivalent to standard consistency, 1-consistency validates internal coherence
within a fixed axiomatic frame.
\end{definition}

\subsection{0-Consistency and Inversion Systems}

Let $I: \mathcal{L} \to \mathcal{L}$ be a domain-specific inversion operator
acting on sentences or structural objects in $S$.

\begin{definition}[Iterated Inversion System]
Given a formal system $S_0 = S$ and an inversion operator $I$, the $n$-th
inversion extension $S_I^n$ is inductively defined as:
\[
S_I^{n+1} = S_I^n \cup \{ I(\varphi) \mid \varphi \in \text{Th}(S_I^n) \}.
\]
The full inductive limit system $S_I^\infty$ is given by:
\[
S_I^\infty = \bigcup_{n \in \mathbb{N}} S_I^n.
\]
\end{definition}

\begin{definition}[0-Consistency]
A system $S$ is \emph{0-consistent} with respect to $I$ (denoted $0\text{C}_I$)
if its inductive limit system $S_I^\infty$ is 1-consistent:
\[
S_I^\infty \nvdash \bot.
\]
\end{definition}

\subsection{The Sufficient Invariant Baseline ($B_0$)}

\begin{definition}[Sufficient Invariant Baseline]
The \emph{Sufficient Invariant Baseline} $B_0$ of a system $S$ under inversion
$I$ is the fixed-point sub-space (or fixed theory):
\[
B_0 \equiv \operatorname{Fix}(I) = \{ x \in \mathcal{L} \mid I(x) = x \}.
\]
\end{definition}

A system is trivially 0-consistent on $B_0$. Systemic friction or non-closure
arises exclusively when operating on elements $x \notin B_0$.

\section{Duality and Non-Closure}
% ============================================================

In classical propositional logic, truth-value assignments 
$0 \mapsto \text{False}, 1 \mapsto \text{True}$ and $0 \mapsto \text{True}, 1 \mapsto \text{False}$
are isomorphic under boolean negation $\neg$. 

However, 0-Consistency goes beyond static boolean relabeling:
\begin{enumerate}
    \item If $I$ is an involution ($I^2 = \text{id}_{\mathcal{L}}$) and $S$ is
    closed under $I$, $0\text{C}$ collapses to $1\text{C}$ on the extended
    frame.
    \item If $S$ is not closed under $I$, $S$ may be 1-consistent while being
    0-inconsistent relative to $I$.
\end{enumerate}

\begin{theorem}[Separation of $1\text{C}$ and $0\text{C}$]
There exist formal systems $S$ and inversion operators $I$ such that $S$ is 
$1\text{C}$ but not $0\text{C}_I$.
\end{theorem}

\begin{proof}
Consider the structure $S = (\mathbb{N}, +)$. $S$ is 1-consistent under Peano
arithmetic. Define $I(a, b) = a - b$. Since $a - b \notin \mathbb{N}$ for 
$a < b$, $S_I^1$ generates elements outside $\mathbb{N}$. Extending $S$ 
naive-syntactically without ring structure updates introduces logical
contradictions with the domain axioms of $\mathbb{N}$. Hence $S$ is 
$1\text{C}$, but $0\text{C}_I$-inconsistent over $\mathbb{N}$.
\end{proof}

\section{Universal Inversion Axiom}
% ============================================================

To allow structural cross-referencing across mathematical sub-disciplines, we
state the operational postulate governing universal self-negation.

\begin{axiom}[Universal Inversion Endofunctor]
Let $\mathbf{Sys}$ denote the category of formal mathematical systems. There
exists an inversion endofunctor $\mathcal{I}: \mathbf{Sys} \to \mathbf{Sys}$
such that for any object $S \in \mathbf{Sys}$:
\begin{enumerate}[label=(\roman*)]
    \item $\mathcal{I}(S)$ maps structural predicates to their dual complements.
    \item $\mathcal{I}^2 \cong \text{id}_{\mathbf{Sys}}$ up to canonical isomorphism.
    \item $S$ is globally 0-consistent if and only if $\mathcal{I}(S)$ preserves
    the satisfiability of $S$.
\end{enumerate}
\end{axiom}
